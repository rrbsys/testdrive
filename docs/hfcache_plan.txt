The error is the key clue:
Unrecognized processing class in google/owlv2-base-patch16-ensemble
This almost always means your installed transformers version is too old to support the OWLv2 processor, not that your plugin is wrong.
The repository google/owlv2-base-patch16-ensemble contains an Owlv2Processor and preprocessor_config.json. Older versions of transformers don't know how to instantiate it, producing exactly the error you're seeing.

First check your version
python -c "import transformers; print(transformers.__version__)"
If it's older than about 4.39–4.40, that's the likely cause.
Upgrade:

pip install -U transformers
or
pip install "transformers>=4.40"
If you must support older transformers
Your load_processor() fallback isn't enough because:
AutoProcessor.from_pretrained(...)
fails for the same reason.
A more compatible approach is to manually construct the processor.

For example:

from transformers import (
    Owlv2Processor,
    Owlv2ImageProcessor,
    CLIPTokenizerFast,
)

image_processor = Owlv2ImageProcessor.from_pretrained(
    repo,
    cache_dir=cd,
)

tokenizer = CLIPTokenizerFast.from_pretrained(
    repo,
    subfolder="tokenizer",
    cache_dir=cd,
)

processor = Owlv2Processor(
    image_processor=image_processor,
    tokenizer=tokenizer,
)
For older releases that don't expose Owlv2ImageProcessor, fall back to:
from transformers import OwlViTImageProcessor
since the preprocessing pipeline is effectively identical.

from pathlib import Path
import shutil

from huggingface_hub import hf_hub_download
from transformers import AutoImageProcessor, AutoProcessor


def _ensure_local_repo(
    repo: str,
    cache_dir: Path,
    required_files: list[str],
) -> Path:
    """
    Download the required Hugging Face files into a normal directory
    (no snapshot symlinks) and return that directory.
    """

    local = cache_dir / "models" / repo.replace("/", "__")
    local.mkdir(parents=True, exist_ok=True)

    missing = [
        f for f in required_files
        if not (local / f).exists()
    ]

    for filename in missing:
        src = hf_hub_download(
            repo_id=repo,
            filename=filename,
            cache_dir=cache_dir,
        )
        shutil.copy2(src, local / filename)

    return local


def load_processor(
    repo: str,
    cd: Path,
    processor_class=None,
    required_files=None,
    **kwargs,
):
    if required_files:
        repo = _ensure_local_repo(repo, cd, required_files)

    attempts = []

    if processor_class:
        attempts.append((
            processor_class.__name__,
            lambda: processor_class.from_pretrained(repo, **kwargs),
        ))

    attempts.extend([
        (
            "AutoProcessor",
            lambda: AutoProcessor.from_pretrained(repo, **kwargs),
        ),
        (
            "AutoImageProcessor",
            lambda: AutoImageProcessor.from_pretrained(repo, **kwargs),
        ),
    ])

    errors = []

    for name, loader in attempts:
        try:
            return loader()
        except Exception as exc:
            errors.append(f"{name}: {exc}")

    raise RuntimeError(
        f"Processor load failed for {repo}:\n"
        + "\n".join(errors)
    )
